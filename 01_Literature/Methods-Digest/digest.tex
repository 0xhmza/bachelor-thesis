\providecommand\pcback[3]{}
\renewcommand\pcback[3]{%
  \ifcsdef{blp@#1@#2}{}{%
    \csgdef{blp@#1@#2}{}%
    \ifcsdef{bl@#1}{\csgappto{bl@#1}{, }}{\csgdef{bl@#1}{}}%
    \csgappto{bl@#1}{\hyperlink{#3}{#2}}}}

% \pc{key}{page}{snippet}: passage citation [n, p. x] -> highlighted passage
\newcommand\pc[3]{%
  \edef\pc@hash{\pdfmdfivesum{\detokenize{#1|#2|#3}}}%
  \ifcsdef{cbk@\pc@hash}{}{%
    \csxdef{cbk@\pc@hash}{}%
    \Hy@raisedlink{\hypertarget{c:\pc@hash}{}}}%
  \edef\pc@tmp{\noexpand\protected@write\noexpand\@auxout{}%
    {\string\pcback{#1}{\noexpand\thepage}{c:\pc@hash}}}\pc@tmp
  \begingroup\hypersetup{linkcolor=LinkBlue}%
  \hyperlink{a:\pc@hash}{[\pnum{#1},\,p.\,#2]}\endgroup}
% \pp{key}: [n] -> first page of the appended full text
