% Outlines drawn by hand: display, stored data, annotation.
\newcommand{\displayshape}[1]{%
  \begin{scope}[on background layer]
    \path[draw=linkblue, line width=0.6pt, fill=linkblue!8]
      ($(#1.west)+(-4mm,0)$) to[out=50,in=180] (#1.north west) --
      ($(#1.north east)+(-1mm,0)$) .. controls +(7mm,0) and +(7mm,0) ..
      ($(#1.south east)+(-1mm,0)$) -- (#1.south west) to[out=180,in=-50]
      ($(#1.west)+(-4mm,0)$) -- cycle;
  \end{scope}}%
\newcommand{\storedshape}[1]{%
  \begin{scope}[on background layer]
    \path[draw=muted, line width=0.6pt, fill=muted!8]
      (#1.north west) -- (#1.north east) to[bend right=40] (#1.south east) --
      (#1.south west) to[bend left=40] (#1.north west) -- cycle;
  \end{scope}}%
\newcommand{\annot}[2]{% #1 annotation node, #2 point annotated
  \draw[muted, line width=0.5pt]
    ($(#1.north west)+(1.6mm,0)$) -- (#1.north west) -- (#1.south west) --
    ($(#1.south west)+(1.6mm,0)$);
  \draw[muted, line width=0.5pt, dash pattern=on 2pt off 1.5pt] (#1.west) -- (#2);}%

% Harm-tree analysis
\node[pre, text width=4.3cm] (p3) at (7.2,-8.85)
